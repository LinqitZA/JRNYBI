-- =============================================================================
-- JRNY ERP — Reporting Schema with Denormalized Views
-- =============================================================================
-- Creates the 'reporting' schema containing denormalized views that join across
-- JRNY's domain schemas for simplified report writing. JRNYBI users query flat
-- views instead of navigating 10+ schemas.
--
-- All views include entity_id and/or branch_id columns for RLS filtering.
-- Views use CREATE OR REPLACE so re-running is safe (idempotent).
--
-- Feature: #2236
-- =============================================================================

BEGIN;

-- ─── Step 1: Create reporting schema ─────────────────────────────────────────
CREATE SCHEMA IF NOT EXISTS reporting;

-- ─── Step 2: Ensure jrnybi_reader role exists ────────────────────────────────
DO $$
BEGIN
    IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = 'jrnybi_reader') THEN
        CREATE ROLE jrnybi_reader WITH
            LOGIN
            PASSWORD 'jrnybi_reader_dev_password'
            NOSUPERUSER
            NOCREATEDB
            NOCREATEROLE;
        RAISE NOTICE 'Created reporting user: jrnybi_reader';
    ELSE
        RAISE NOTICE 'Reporting user jrnybi_reader already exists';
    END IF;
END
$$;

GRANT USAGE ON SCHEMA reporting TO jrnybi_reader;

-- ─── View 1: v_sales_orders ──────────────────────────────────────────────────
-- Denormalized sales orders with line items, customer details, and product info.
-- Grain: one row per sales order line.
CREATE OR REPLACE VIEW reporting.v_sales_orders AS
SELECT
    so.id                       AS order_id,
    so.order_number,
    so.order_date,
    so.status                   AS order_status,
    so.customer_id,
    so.customer_name,
    c.code                      AS customer_code,
    c.email                     AS customer_email,
    c.phone                     AS customer_phone,
    c.payment_terms_days        AS customer_payment_terms,
    so.sales_rep_id,
    so.sales_rep_name,
    sol.id                      AS line_id,
    sol.line_number,
    sol.product_id,
    sol.product_code,
    sol.product_name,
    p.category                  AS product_category,
    p.uom                       AS product_uom,
    sol.quantity,
    sol.unit_price,
    sol.cost_price,
    sol.gp_margin               AS line_gp_margin,
    sol.vat_rate                AS line_vat_rate,
    sol.vat_amount              AS line_vat_amount,
    sol.line_total,
    sol.line_total_incl_vat,
    sol.is_price_override,
    so.subtotal                 AS order_subtotal,
    so.vat_total                AS order_vat_total,
    so.total                    AS order_total,
    so.gp_margin                AS order_gp_margin,
    so.currency,
    so.source_quotation_id,
    so.source_quotation_number,
    so.credit_check_passed,
    so.confirmed_at,
    so.confirmed_by,
    so.created_at,
    so.updated_at,
    so.branch_id,
    b.name                      AS branch_name,
    b.code                      AS branch_code,
    c.entity_id
FROM sales.sales_orders so
LEFT JOIN sales.sales_order_lines sol ON sol.order_id = so.id
LEFT JOIN crm.customers c ON c.id = so.customer_id
LEFT JOIN inventory.products p ON p.id = sol.product_id
LEFT JOIN core.branches b ON b.id = so.branch_id;

COMMENT ON VIEW reporting.v_sales_orders IS
'Denormalized sales orders at line-item grain. Joins sales_orders, sales_order_lines, customers, products, branches. Use for sales analysis, product mix, margin analysis. Filter by entity_id, branch_id, order_date, order_status.';

-- Denormalized invoices with line items, customer details, and linked SO.
-- Grain: one row per invoice line.

SELECT

    inv.invoice_number,
    inv.invoice_date,
    inv.due_date,
    inv.status                  AS invoice_status,
    inv.customer_id,
    inv.customer_name,
    c.code                      AS customer_code,

    c.payment_terms_days        AS customer_payment_terms,
    il.id                       AS line_id,
    il.line_number,
    il.description              AS line_description,
    il.quantity,
    il.unit_price,
    il.vat_rate                 AS line_vat_rate,
    il.vat_amount               AS line_vat_amount,
    il.line_total,
    il.line_total_incl_vat,
    il.tax_code_id,
    il.tax_code,
    inv.subtotal                AS invoice_subtotal,
    inv.vat_total               AS invoice_vat_total,
    inv.total                   AS invoice_total,
    inv.amount_paid,
    inv.balance                 AS invoice_balance,

-- ─── View 3: v_inventory_levels ──────────────────────────────────────────────
-- Current stock levels by product and warehouse with product details.
-- Grain: one row per product per warehouse.
CREATE OR REPLACE VIEW reporting.v_inventory_levels AS
SELECT
    soh.id                      AS stock_id,
    soh.product_id,
    p.code                      AS product_code,
    p.name                      AS product_name,
    p.category                  AS product_category,
    p.uom                       AS product_uom,
    p.cost_price,
    p.selling_price,
    p.reorder_level             AS reorder_point,
    p.reorder_quantity,
    p.is_active                 AS product_is_active,
    soh.warehouse_id,
    w.code                      AS warehouse_code,
    w.name                      AS warehouse_name,
    w.warehouse_type,
    soh.quantity_on_hand,
    soh.quantity_reserved,
    soh.quantity_available,
    CASE
        WHEN p.reorder_level IS NOT NULL AND soh.quantity_available <= p.reorder_level
        THEN true ELSE false
    END                         AS below_reorder_point,
    (soh.quantity_on_hand * COALESCE(p.cost_price, 0))  AS stock_value_at_cost,
    (soh.quantity_on_hand * COALESCE(p.selling_price, 0)) AS stock_value_at_sell,
    soh.last_updated,
    soh.created_at,
    w.branch_id,
    b.name                      AS branch_name,
    b.code                      AS branch_code,
    w.branch_id                 AS entity_id  -- branches represent entities in JRNY
FROM inventory.stock_on_hand soh
JOIN inventory.products p ON p.id = soh.product_id
JOIN inventory.warehouses w ON w.id = soh.warehouse_id
LEFT JOIN core.branches b ON b.id = w.branch_id;

COMMENT ON VIEW reporting.v_inventory_levels IS
'Current stock levels per product per warehouse. Joins stock_on_hand, products, warehouses, branches. Includes reorder alerts, stock valuation at cost and selling price. Filter by branch_id, warehouse_id, product_category, below_reorder_point.';

    sl.description,
    sl.reference,

-- ─── View 6: v_general_ledger ────────────────────────────────────────────────
-- Journal entries with account details and batch context.
-- Grain: one row per journal line.
CREATE OR REPLACE VIEW reporting.v_general_ledger AS
SELECT
    jl.id                       AS line_id,
    jl.batch_id,
    jb.batch_number,
    jb.description              AS batch_description,
    jb.posting_date             AS entry_date,
    jb.status                   AS batch_status,
    jb.reference                AS source_document,
    jl.line_number,
    jl.account_id,
    ga.code                     AS account_code,
    ga.name                     AS account_name,
    ga.account_type,
    ga.account_group,
    ga.ifrs18_category,
    jl.description              AS line_description,
    jl.debit_oc                  AS debit,
    jl.credit_oc                 AS credit,
    (COALESCE(jl.debit_oc, 0) - COALESCE(jl.credit_oc, 0)) AS net_amount,
    jl.dim_entity_id,
    jl.dim_function_id,
    jl.dim_project_id,
    jb.currency,
    jb.is_intercompany,
    jb.destination_entity_id,

-- ─── View 7: v_customers ────────────────────────────────────────────────────
-- Customer master data with aggregated sales totals, last order date,
-- outstanding balance.
-- Grain: one row per customer.
CREATE OR REPLACE VIEW reporting.v_customers AS
SELECT
    c.id                        AS customer_id,
    c.code                      AS customer_code,
    c.name                      AS customer_name,
    c.trading_name,
    c.email,
    c.phone,
    c.vat_number,
    c.registration_number,
    c.billing_address,
    c.delivery_address,
    c.credit_limit,
    c.payment_terms_days,
    c.currency,
    c.customer_type,
    c.contact_name,
    c.contact_email,
    c.contact_phone,
    c.is_active,
    c.price_list_id,
    c.invoicing_frequency,
    c.created_at,
    c.updated_at,
    c.entity_id,
    -- Aggregated sales metrics
    COALESCE(so_agg.total_orders, 0)        AS total_orders,
    COALESCE(so_agg.total_order_value, 0)   AS total_order_value,
    so_agg.last_order_date,
    -- Aggregated invoice / outstanding balance
    COALESCE(inv_agg.total_invoices, 0)     AS total_invoices,
    COALESCE(inv_agg.total_invoiced, 0)     AS total_invoiced,
    COALESCE(inv_agg.total_paid, 0)         AS total_paid,
    COALESCE(inv_agg.outstanding_balance, 0) AS outstanding_balance,
    inv_agg.last_invoice_date,
    -- Derive branch_id from entity_id
    c.entity_id                              AS branch_id
FROM crm.customers c
LEFT JOIN (
    SELECT
        customer_id,
        COUNT(*)            AS total_orders,
        SUM(total)          AS total_order_value,
        MAX(order_date)     AS last_order_date
    FROM sales.sales_orders
    WHERE status NOT IN ('cancelled', 'draft')
    GROUP BY customer_id
) so_agg ON so_agg.customer_id = c.id
LEFT JOIN (
    SELECT
        customer_id,
        COUNT(*)            AS total_invoices,
        SUM(total)          AS total_invoiced,
        SUM(amount_paid)    AS total_paid,
        SUM(balance)        AS outstanding_balance,
        MAX(invoice_date)   AS last_invoice_date
    FROM finance.invoices
    WHERE status NOT IN ('cancelled', 'draft')
    GROUP BY customer_id
) inv_agg ON inv_agg.customer_id = c.id
WHERE c.deleted_at IS NULL;

COMMENT ON VIEW reporting.v_customers IS
'Customer master with aggregated sales and invoice totals. Joins customers, sales_orders (aggregated), invoices (aggregated). Excludes soft-deleted customers. Use for customer analysis, credit management, segmentation. Filter by entity_id, is_active, customer_type.';


-- ─── View 8: v_suppliers ─────────────────────────────────────────────────────
-- Supplier master data with aggregated PO totals and last PO date.
-- Grain: one row per supplier.
CREATE OR REPLACE VIEW reporting.v_suppliers AS
SELECT
    v.id                        AS supplier_id,
    v.code                      AS supplier_code,
    v.name                      AS supplier_name,
    v.trading_name,
    v.email,
    v.phone,
    v.vat_number,
    v.registration_number,
    v.address,
    v.bank_name,
    v.bank_account_number,
    v.bank_branch_code,
    v.bank_account_type,
    v.payment_terms_days,
    v.currency,
    v.contact_name,
    v.contact_email,
    v.contact_phone,
    v.categories                AS supplier_categories,
    v.is_active,
    v.created_at,
    v.updated_at,
    -- Aggregated PO metrics
    COALESCE(po_agg.total_pos, 0)       AS total_purchase_orders,
    COALESCE(po_agg.total_po_value, 0)  AS total_po_value,
    po_agg.last_po_date,
    po_agg.avg_delivery_lead_days,
    -- Suppliers don't have entity_id; derive from POs
    po_agg.primary_entity_id             AS entity_id,
    po_agg.primary_entity_id             AS branch_id
FROM crm.vendors v
LEFT JOIN (
    SELECT
        vendor_id,
        COUNT(*)                            AS total_pos,
        SUM(total)                          AS total_po_value,
        MAX(po_date)                        AS last_po_date,
        AVG(EXTRACT(EPOCH FROM (
            COALESCE(approved_at, created_at) - created_at
        )) / 86400)::numeric(10,1)          AS avg_delivery_lead_days,
        -- Most common entity for this vendor
        (ARRAY_AGG(entity_id ORDER BY po_date DESC))[1] AS primary_entity_id
    FROM procurement.purchase_orders
    WHERE status NOT IN ('cancelled', 'draft')
    GROUP BY vendor_id
) po_agg ON po_agg.vendor_id = v.id
WHERE v.deleted_at IS NULL;

COMMENT ON VIEW reporting.v_suppliers IS
'Supplier master with aggregated purchase order totals. Joins vendors, purchase_orders (aggregated). Excludes soft-deleted vendors. Use for vendor analysis, spend management, supplier performance. Filter by is_active, entity_id, supplier_categories.';


-- ─── View 9: v_product_catalogue ─────────────────────────────────────────────
-- Product master with category, stock summary across all warehouses, and
-- price list entries.
-- Grain: one row per product.
CREATE OR REPLACE VIEW reporting.v_product_catalogue AS
SELECT
    p.id                        AS product_id,
    p.code                      AS product_code,
    p.name                      AS product_name,
    p.short_description,
    p.description               AS product_description,
    p.category,
    pg.name                     AS product_group_name,
    p.uom,
    p.cost_price,
    p.selling_price,
    p.vat_code,
    p.barcode,
    p.tracking_type,
    p.stock_tracking,
    p.reorder_level,
    p.reorder_quantity,
    p.is_active,
    p.created_at,
    p.updated_at,
    -- Aggregated stock across all warehouses
    COALESCE(stk.total_on_hand, 0)      AS total_qty_on_hand,
    COALESCE(stk.total_reserved, 0)     AS total_qty_reserved,
    COALESCE(stk.total_available, 0)    AS total_qty_available,
    stk.warehouse_count,
    COALESCE(stk.total_on_hand * COALESCE(p.cost_price, 0), 0) AS total_stock_value_at_cost,
    -- Price list tiers
    pl_agg.price_lists,
    -- Products are global; entity_id/branch_id included for RLS filter consistency
    CAST(NULL AS uuid)                  AS entity_id,
    CAST(NULL AS uuid)                  AS branch_id
FROM inventory.products p
LEFT JOIN inventory.product_groups pg ON pg.id::text = p.category
LEFT JOIN (
    SELECT
        product_id,
        SUM(quantity_on_hand)   AS total_on_hand,
        SUM(quantity_reserved)  AS total_reserved,
        SUM(quantity_available) AS total_available,
        COUNT(*)                AS warehouse_count
    FROM inventory.stock_on_hand
    GROUP BY product_id
) stk ON stk.product_id = p.id
LEFT JOIN (
    SELECT
        pll.product_id,
        jsonb_agg(jsonb_build_object(
            'price_list', pl.name,
            'currency', pl.currency,
            'price_type', pll.price_type,
            'value', pll.value,
            'effective_from', pl.effective_from,
            'effective_to', pl.effective_to
        ) ORDER BY pl.name) AS price_lists
    FROM pricing.price_list_lines pll
    JOIN pricing.price_lists pl ON pl.id = pll.price_list_id
    WHERE pl.is_active = true
    GROUP BY pll.product_id
) pl_agg ON pl_agg.product_id = p.id;

COMMENT ON VIEW reporting.v_product_catalogue IS
'Product catalogue with category, aggregated stock levels, and active price list tiers. Joins products, product_groups, stock_on_hand (aggregated), price_list_lines. entity_id/branch_id are NULL (products are global). Use for product analysis, pricing review, stock valuation. Filter by category, is_active, stock_tracking.';


-- ─── View 10: v_otif ─────────────────────────────────────────────────────────
-- On-Time In-Full (OTIF) delivery performance per PO line.
-- Measures whether each ordered item was received by the expected delivery date
-- AND in the full quantity ordered.
-- Grain: one row per PO line item (with GRN receipt data aggregated).
CREATE OR REPLACE VIEW reporting.v_otif AS
SELECT
    po.id                                   AS po_id,
    po.po_number,
    po.po_date,
    po.expected_delivery_date,
    po.status                               AS po_status,
    po.vendor_id,
    po.vendor_name,
    poi.id                                  AS po_line_id,
    poi.line_number,
    poi.product_id,
    poi.product_code,
    poi.product_name,
    poi.quantity                             AS ordered_quantity,
    COALESCE(grn_agg.total_received_qty, 0) AS received_quantity,
    grn_agg.first_received_date,
    grn_agg.last_received_date,
    -- On-Time: received (last receipt) by expected delivery date
    CASE
        WHEN grn_agg.last_received_date IS NOT NULL
             AND po.expected_delivery_date IS NOT NULL
             AND grn_agg.last_received_date <= po.expected_delivery_date
        THEN true
        ELSE false
    END                                     AS is_on_time,
    -- In-Full: total received >= ordered quantity
    CASE
        WHEN COALESCE(grn_agg.total_received_qty, 0) >= poi.quantity
        THEN true
        ELSE false
    END                                     AS is_in_full,
    -- OTIF: both on-time AND in-full
    CASE
        WHEN grn_agg.last_received_date IS NOT NULL
             AND po.expected_delivery_date IS NOT NULL
             AND grn_agg.last_received_date <= po.expected_delivery_date
             AND COALESCE(grn_agg.total_received_qty, 0) >= poi.quantity
        THEN true
        ELSE false
    END                                     AS is_otif,
    po.entity_id
FROM procurement.purchase_orders po
JOIN procurement.purchase_order_items poi ON poi.purchase_order_id = po.id
LEFT JOIN (
    SELECT
        grn.po_id,
        grl.product_id,
        SUM(grl.received_quantity)   AS total_received_qty,
        MIN(grn.received_date)       AS first_received_date,
        MAX(grn.received_date)       AS last_received_date
    FROM inventory.grn_headers grn
    JOIN inventory.grn_lines grl ON grl.grn_id = grn.id
    GROUP BY grn.po_id, grl.product_id
) grn_agg ON grn_agg.po_id = po.id AND grn_agg.product_id = poi.product_id;

COMMENT ON VIEW reporting.v_otif IS
'On-Time In-Full (OTIF) delivery performance per PO line item. Joins purchase_orders, purchase_order_items, grn_headers, grn_lines. OTIF = received by expected_delivery_date AND received_qty >= ordered_qty. Use for supplier delivery performance, procurement KPIs. Filter by entity_id, po_date, vendor_id, is_otif.';

GRANT SELECT ON reporting.v_otif TO jrnybi_reader;


-- ─── View 11: v_open_po_aging ────────────────────────────────────────────────
-- Open purchase orders with aging buckets based on days overdue.
-- Only includes POs in active/open statuses (Sent, Partial, Confirmed).
-- Grain: one row per open PO.
CREATE OR REPLACE VIEW reporting.v_open_po_aging AS
SELECT
    po.id                                   AS po_id,
    po.po_number,
    po.po_date,
    po.expected_delivery_date,
    po.status                               AS po_status,
    po.vendor_id,
    v.name                                  AS supplier_name,
    v.code                                  AS supplier_code,
    po.total                                AS po_total,
    po.currency,
    -- Days overdue (negative = not yet due)
    (CURRENT_DATE - po.expected_delivery_date) AS days_overdue,
    -- Aging buckets
    CASE
        WHEN po.expected_delivery_date IS NULL THEN 'No Date'
        WHEN CURRENT_DATE - po.expected_delivery_date <= 0 THEN 'Not Due'
        WHEN CURRENT_DATE - po.expected_delivery_date BETWEEN 1 AND 7 THEN '1-7 Days'
        WHEN CURRENT_DATE - po.expected_delivery_date BETWEEN 8 AND 14 THEN '8-14 Days'
        WHEN CURRENT_DATE - po.expected_delivery_date BETWEEN 15 AND 30 THEN '15-30 Days'
        WHEN CURRENT_DATE - po.expected_delivery_date BETWEEN 31 AND 60 THEN '31-60 Days'
        WHEN CURRENT_DATE - po.expected_delivery_date BETWEEN 61 AND 90 THEN '61-90 Days'
        ELSE '90+ Days'
    END                                     AS aging_bucket,

-- ─── View 22: v_budget_variance ─────────────────────────────────────────────
-- Budget vs actual variance analysis by GL account and period.
-- Grain: one row per budget allocation (GL account + period + entity).
CREATE OR REPLACE VIEW reporting.v_budget_variance AS
SELECT
    ba.id                        AS budget_allocation_id,
    ba.entity_id,
    ba.gl_account_id,
    ga.code                      AS gl_account_code,
    ga.name                      AS gl_account_name,
    ga.account_type,
    ga.account_group,
    ag.name                      AS account_group_name,
    ba.fiscal_year_id,
    fy.fiscal_year_label,
    fy.start_date                AS fy_start_date,
    fy.end_date                  AS fy_end_date,
    ba.period_id,
    ap.name                      AS period_name,
    ap.code                      AS period_code,
    ap.period_number,
    ap.start_date                AS period_start_date,
    ap.end_date                  AS period_end_date,
    ba.budget_amount,
    ba.budget_currency,
    ba.committed_spend,
    ba.actual_spend              AS recorded_actual_spend,
    -- Calculated actuals from journal lines (using debit_oc/credit_oc)
    COALESCE(actuals.actual_debit_oc, 0)  AS actual_debit_oc,
    COALESCE(actuals.actual_credit_oc, 0) AS actual_credit_oc,
    COALESCE(actuals.actual_debit_oc, 0) - COALESCE(actuals.actual_credit_oc, 0) AS actual_net_oc,
    -- Variance = budget - actual net
    ba.budget_amount - (COALESCE(actuals.actual_debit_oc, 0) - COALESCE(actuals.actual_credit_oc, 0)) AS variance_amount,
    -- Variance percentage (avoid division by zero)
    CASE
        WHEN ba.budget_amount IS NOT NULL AND ba.budget_amount != 0
        THEN ((ba.budget_amount - (COALESCE(actuals.actual_debit_oc, 0) - COALESCE(actuals.actual_credit_oc, 0))) / ba.budget_amount * 100)
        ELSE NULL
    END                          AS variance_percentage,
    ba.description,
    ba.is_active,
    ba.created_at,
    ba.updated_at
FROM finance.budget_allocations ba
JOIN finance.gl_accounts ga ON ga.id = ba.gl_account_id
LEFT JOIN finance.account_groups ag ON ag.code = ga.account_group AND ag.entity_id = ba.entity_id
JOIN finance.fiscal_years fy ON fy.id = ba.fiscal_year_id
JOIN finance.accounting_periods ap ON ap.id = ba.period_id
LEFT JOIN (
    SELECT
        jl.account_id,
        jl.entity_id,
        jb.period_id,
        SUM(COALESCE(jl.debit_oc, 0))  AS actual_debit_oc,
        SUM(COALESCE(jl.credit_oc, 0)) AS actual_credit_oc
    FROM finance.journal_lines jl
    JOIN finance.journal_batches jb ON jb.id = jl.batch_id
    WHERE jb.status = 'Posted'
    GROUP BY jl.account_id, jl.entity_id, jb.period_id
) actuals ON actuals.account_id = ba.gl_account_id
         AND actuals.entity_id = ba.entity_id
         AND actuals.period_id = ba.period_id;

COMMENT ON VIEW reporting.v_budget_variance IS
'Budget vs actual variance by GL account and period. Joins budget_allocations, gl_accounts, account_groups, fiscal_years, accounting_periods, journal_lines (actuals from debit_oc/credit_oc). Use for budget monitoring, variance analysis, departmental spending. Filter by entity_id, fiscal_year_label, period_name, account_type, variance_percentage.';

GRANT SELECT ON reporting.v_budget_variance TO jrnybi_reader;

    inv.branch_id,
    -- Payment aggregation
    COALESCE(pay_agg.total_payments, 0)  AS total_payments,
    COALESCE(pay_agg.total_paid, 0)      AS total_payment_amount,
    pay_agg.first_payment_date,
    pay_agg.last_payment_date,
    -- Payment compliance metrics
    CASE
        WHEN pay_agg.last_payment_date IS NOT NULL
        THEN (pay_agg.last_payment_date - inv.due_date)
        ELSE NULL
    END                          AS days_late,
    CASE
        WHEN inv.balance <= 0 THEN 'Paid'
        WHEN inv.due_date >= CURRENT_DATE THEN 'Not Due'
        WHEN (CURRENT_DATE - inv.due_date) BETWEEN 1 AND 30 THEN 'Overdue 1-30'
        WHEN (CURRENT_DATE - inv.due_date) BETWEEN 31 AND 60 THEN 'Overdue 31-60'
        WHEN (CURRENT_DATE - inv.due_date) BETWEEN 61 AND 90 THEN 'Overdue 61-90'
        ELSE 'Overdue 90+'
    END                          AS compliance_status,
    CASE
        WHEN inv.balance <= 0 AND pay_agg.last_payment_date IS NOT NULL
             AND pay_agg.last_payment_date <= inv.due_date
        THEN true
        ELSE false
    END                          AS paid_on_time,
    inv.posted_at,
    inv.created_at,
    inv.updated_at
FROM finance.invoices inv
LEFT JOIN crm.customers c ON c.id = inv.customer_id
LEFT JOIN crm.customer_groups cg ON cg.id = c.customer_group_id
LEFT JOIN (
    SELECT
        invoice_id,
        COUNT(*)              AS total_payments,
        SUM(amount)           AS total_paid,
        MIN(payment_date)     AS first_payment_date,
        MAX(payment_date)     AS last_payment_date
    FROM finance.customer_payments
    WHERE status != 'voided'
    GROUP BY invoice_id
) pay_agg ON pay_agg.invoice_id = inv.id
WHERE inv.status NOT IN ('cancelled', 'draft');

COMMENT ON VIEW reporting.v_payment_compliance IS
'Payment compliance analysis per invoice with customer group info. Joins invoices, customers, customer_groups, customer_payments (aggregated). Excludes cancelled/draft invoices. Tracks payment timing, on-time status, overdue buckets. Filter by entity_id, branch_id, compliance_status, customer_group_name, paid_on_time.';

-- ─── View 25: v_inventory_aging ─────────────────────────────────────────────
-- Inventory aging by product and warehouse with aging buckets based on last receipt.
-- Grain: one row per product per warehouse (where quantity > 0).
CREATE OR REPLACE VIEW reporting.v_inventory_aging AS
SELECT
    soh.id                       AS stock_id,
    soh.product_id,
    p.code                       AS product_code,
    p.name                       AS product_name,
    pg.name                      AS category,
    p.uom                        AS product_uom,
    soh.warehouse_id,
    w.code                       AS warehouse_code,
    w.name                       AS warehouse_name,
    soh.quantity_on_hand,
    soh.quantity_reserved,
    soh.quantity_available,
    -- Latest unit cost from valuation snapshots
    vs.unit_cost                 AS latest_unit_cost,
    vs.costing_method,
    vs.snapshot_date             AS valuation_date,
    (soh.quantity_on_hand * COALESCE(vs.unit_cost, p.cost_price, 0)) AS stock_value,
    -- Last receipt date from stock movements
    lr.last_receipt_date,
    CASE
        WHEN lr.last_receipt_date IS NULL THEN 'No Receipt'
        WHEN (CURRENT_DATE - lr.last_receipt_date::date) <= 30 THEN '0-30 Days'
        WHEN (CURRENT_DATE - lr.last_receipt_date::date) <= 60 THEN '31-60 Days'
        WHEN (CURRENT_DATE - lr.last_receipt_date::date) <= 90 THEN '61-90 Days'
        WHEN (CURRENT_DATE - lr.last_receipt_date::date) <= 180 THEN '91-180 Days'
        WHEN (CURRENT_DATE - lr.last_receipt_date::date) <= 365 THEN '181-365 Days'
        ELSE '365+ Days'
    END                          AS aging_bucket,
    CASE
        WHEN lr.last_receipt_date IS NOT NULL
        THEN (CURRENT_DATE - lr.last_receipt_date::date)
        ELSE NULL
    END                          AS days_since_receipt,
    w.branch_id,
    w.branch_id                  AS entity_id
FROM inventory.stock_on_hand soh
JOIN inventory.products p ON p.id = soh.product_id
LEFT JOIN inventory.product_groups pg ON pg.id = p.product_group_id
JOIN inventory.warehouses w ON w.id = soh.warehouse_id
LEFT JOIN LATERAL (
    SELECT vs2.unit_cost, vs2.costing_method, vs2.snapshot_date
    FROM inventory.valuation_snapshots vs2
    WHERE vs2.product_id = soh.product_id AND vs2.warehouse_id = soh.warehouse_id
    ORDER BY vs2.snapshot_date DESC
    LIMIT 1
) vs ON true
LEFT JOIN LATERAL (
    SELECT MAX(sm.transaction_date) AS last_receipt_date
    FROM inventory.stock_movements sm
    WHERE sm.product_id = soh.product_id
      AND sm.warehouse_id = soh.warehouse_id
      AND sm.movement_type IN ('GRN_RECEIPT', 'TRANSFER_IN')
) lr ON true
WHERE soh.quantity_on_hand > 0;

COMMENT ON VIEW reporting.v_inventory_aging IS
'Inventory aging by product and warehouse with aging buckets based on last receipt date. Joins stock_on_hand, products, product_groups, warehouses, valuation_snapshots, stock_movements. Use for dead stock identification, inventory health monitoring. Filter by entity_id, aging_bucket, category, warehouse_code.';

-- ─── View 27: v_slow_moving_stock ───────────────────────────────────────────
-- Slow-moving stock (no outbound movement in 60+ days).
-- Grain: one row per product per warehouse (where quantity > 0 and no recent outbound).
CREATE OR REPLACE VIEW reporting.v_slow_moving_stock AS
SELECT
    soh.id                       AS stock_id,
    soh.product_id,
    p.code                       AS product_code,
    p.name                       AS product_name,
    pg.name                      AS category,
    p.uom                        AS product_uom,
    soh.warehouse_id,
    w.code                       AS warehouse_code,
    w.name                       AS warehouse_name,
    soh.quantity_on_hand,
    soh.quantity_available,
    (soh.quantity_on_hand * COALESCE(p.cost_price, 0)) AS stock_value_at_cost,
    -- Last outbound movement
    lo.last_outbound_date,
    CASE
        WHEN lo.last_outbound_date IS NOT NULL
        THEN (CURRENT_DATE - lo.last_outbound_date::date)
        ELSE NULL
    END                          AS days_since_outbound,
    CASE
        WHEN lo.last_outbound_date IS NULL THEN 'No Movement'
        WHEN (CURRENT_DATE - lo.last_outbound_date::date) BETWEEN 61 AND 90 THEN '61-90 Days'
        WHEN (CURRENT_DATE - lo.last_outbound_date::date) BETWEEN 91 AND 180 THEN '91-180 Days'
        WHEN (CURRENT_DATE - lo.last_outbound_date::date) BETWEEN 181 AND 365 THEN '181-365 Days'
        ELSE '365+ Days'
    END                          AS slow_moving_bucket,
    w.branch_id,
    w.branch_id                  AS entity_id
FROM inventory.stock_on_hand soh
JOIN inventory.products p ON p.id = soh.product_id
LEFT JOIN inventory.product_groups pg ON pg.id = p.product_group_id
JOIN inventory.warehouses w ON w.id = soh.warehouse_id
LEFT JOIN LATERAL (
    SELECT MAX(sm.transaction_date) AS last_outbound_date
    FROM inventory.stock_movements sm
    WHERE sm.product_id = soh.product_id
      AND sm.warehouse_id = soh.warehouse_id
      AND sm.movement_type IN ('SALE', 'TRANSFER_OUT', 'ADJUSTMENT')
) lo ON true
WHERE soh.quantity_on_hand > 0
  AND (
    lo.last_outbound_date IS NULL
    OR (CURRENT_DATE - lo.last_outbound_date::date) > 60
  );

-- ─── View 28: v_pick_pack_performance ───────────────────────────────────────
-- Pick and pack performance metrics from sales schema pick instructions.
-- Grain: one row per pick instruction.
CREATE OR REPLACE VIEW reporting.v_pick_pack_performance AS
SELECT
    pi.id                        AS pick_instruction_id,
    pi.pick_number,
    pi.sales_order_id,
    pi.sales_order_number,
    pi.customer_name,
    pi.warehouse_id,
    pi.warehouse_name,
    pi.status                    AS pick_status,
    pi.total_lines,
    pi.total_quantity,
    pi.created_at                AS pick_created_at,
    pi.completed_at              AS pick_completed_at,
    pi.created_by                AS pick_created_by,
    pi.completed_by              AS pick_completed_by,
    -- Pick performance metrics
    CASE
        WHEN pi.completed_at IS NOT NULL AND pi.created_at IS NOT NULL
        THEN EXTRACT(EPOCH FROM (pi.completed_at - pi.created_at)) / 60.0
        ELSE NULL
    END                          AS pick_duration_minutes,
    pi.picking_duration_minutes  AS recorded_pick_duration,
    -- Lines performance
    COALESCE(line_agg.total_lines_picked, 0) AS total_lines_picked,
    COALESCE(line_agg.total_qty_picked, 0)   AS total_qty_picked,
    -- Pack performance (via dispatch notes -> cartons)
    dn.id                        AS dispatch_id,
    dn.dispatch_number,
    dn.pack_completed_at,
    COALESCE(carton_agg.total_cartons, 0)    AS total_cartons,
    CASE
        WHEN dn.pack_completed_at IS NOT NULL AND pi.completed_at IS NOT NULL
        THEN EXTRACT(EPOCH FROM (dn.pack_completed_at - pi.completed_at)) / 60.0
        ELSE NULL
    END                          AS pack_duration_minutes,
    pi.entity_id,
    pi.branch_id
FROM sales.pick_instructions pi
LEFT JOIN (
    SELECT
        pil.pick_instruction_id,
        COUNT(*)                        AS total_lines_picked,
        SUM(pil.quantity_picked)         AS total_qty_picked
    FROM sales.pick_instruction_lines pil
    WHERE pil.status = 'Picked'
    GROUP BY pil.pick_instruction_id
) line_agg ON line_agg.pick_instruction_id = pi.id
LEFT JOIN sales.dispatch_notes dn ON dn.pick_instruction_id = pi.id
LEFT JOIN (
    SELECT
        c.dispatch_id,
        COUNT(*)                         AS total_cartons
    FROM inventory.cartons c
    GROUP BY c.dispatch_id
) carton_agg ON carton_agg.dispatch_id = dn.id;

COMMENT ON VIEW reporting.v_pick_pack_performance IS
'Pick and pack performance metrics. Joins pick_instructions, pick_instruction_lines, dispatch_notes, cartons. Tracks pick duration, pack duration, carton counts. Use for warehouse productivity, SLA monitoring, bottleneck identification. Filter by entity_id, branch_id, pick_status, warehouse_id.';

-- ─── View 31: v_bin_utilization ─────────────────────────────────────────────
-- Bin utilization across warehouses with capacity analysis.
-- Grain: one row per bin.
CREATE OR REPLACE VIEW reporting.v_bin_utilization AS
SELECT
    bin.id                       AS bin_id,
    bin.code                     AS bin_code,
    bin.name                     AS bin_name,
    bin.bin_type,
    bin.max_capacity,
    bin.capacity_unit,
    bin.is_active                AS bin_is_active,
    bin.location_id,
    loc.code                     AS location_code,
    loc.name                     AS location_name,
    loc.location_type,
    loc.warehouse_id,
    w.code                       AS warehouse_code,
    w.name                       AS warehouse_name,
    -- Bin stock aggregation
    COALESCE(bs_agg.total_products, 0)  AS total_products,
    COALESCE(bs_agg.total_quantity, 0)  AS total_quantity,
    -- Utilization percentage (if max_capacity is set)
    CASE
        WHEN bin.max_capacity IS NOT NULL AND bin.max_capacity > 0
        THEN (COALESCE(bs_agg.total_quantity, 0) / bin.max_capacity * 100)
        ELSE NULL
    END                          AS utilization_percentage,
    CASE
        WHEN COALESCE(bs_agg.total_quantity, 0) = 0 THEN 'Empty'
        WHEN bin.max_capacity IS NOT NULL AND bin.max_capacity > 0
             AND (COALESCE(bs_agg.total_quantity, 0) / bin.max_capacity * 100) >= 90
        THEN 'Near Full'
        WHEN bin.max_capacity IS NOT NULL AND bin.max_capacity > 0
             AND (COALESCE(bs_agg.total_quantity, 0) / bin.max_capacity * 100) >= 50
        THEN 'Moderate'
        ELSE 'Low'
    END                          AS utilization_status,
    bin.created_at,
    w.branch_id,
    w.branch_id                  AS entity_id
FROM inventory.bins bin
JOIN inventory.locations loc ON loc.id = bin.location_id
JOIN inventory.warehouses w ON w.id = loc.warehouse_id
LEFT JOIN (
    SELECT
        bs.bin_id,
        COUNT(DISTINCT bs.product_id) AS total_products,
        SUM(bs.quantity)              AS total_quantity
    FROM inventory.bin_stock bs
    GROUP BY bs.bin_id
) bs_agg ON bs_agg.bin_id = bin.id;

COMMENT ON VIEW reporting.v_bin_utilization IS
'Bin utilization across warehouses. Joins bins, locations, warehouses, bin_stock. Shows capacity usage, utilization status. Use for warehouse space planning, bin allocation optimization. Filter by entity_id, warehouse_code, utilization_status, bin_type, location_name.';

-- Products at or below reorder level with last supplier info.

    p.uom                        AS product_uom,

    soh.quantity_reserved,

    p.reorder_quantity,
    p.cost_price,
    -- Recommended order value
    (COALESCE(p.reorder_quantity, 0) * COALESCE(p.cost_price, 0)) AS reorder_cost,
    -- Shortfall below reorder level
    (p.reorder_level - soh.quantity_on_hand) AS quantity_shortfall,
    -- Last supplier info
    ls.last_supplier_name,
    ls.last_supplier_code,
    ls.last_po_date,
    ls.last_unit_cost,
    w.branch_id,
    w.branch_id                  AS entity_id
FROM inventory.stock_on_hand soh
JOIN inventory.products p ON p.id = soh.product_id
LEFT JOIN inventory.product_groups pg ON pg.id = p.product_group_id
JOIN inventory.warehouses w ON w.id = soh.warehouse_id
LEFT JOIN LATERAL (
    SELECT
        v.name  AS last_supplier_name,
        v.code  AS last_supplier_code,
        po.po_date AS last_po_date,
        poi.unit_cost_tc AS last_unit_cost
    FROM procurement.purchase_order_items poi
    JOIN procurement.purchase_orders po ON po.id = poi.purchase_order_id
    LEFT JOIN crm.vendors v ON v.id = po.vendor_id
    WHERE poi.product_id = soh.product_id
    ORDER BY po.po_date DESC
    LIMIT 1
) ls ON true
WHERE p.reorder_level IS NOT NULL
  AND p.reorder_level > 0
  AND soh.quantity_on_hand <= p.reorder_level;

'Products at or below reorder level with supplier info. Joins stock_on_hand, products, product_groups, warehouses, purchase_order_items, purchase_orders, vendors. Use for purchasing decisions, stock replenishment, automated reorder alerts. Filter by entity_id, warehouse_code, category.';
